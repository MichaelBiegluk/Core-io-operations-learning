I created a small, responsive calculator app using HTML, CSS, and JavaScript. Open index.html in your browser to run it. The calculator supports digits, decimal point, + − × ÷, clear, backspace, equals, and basic keyboard input (numbers, + - * /, Enter, Backspace, Escape).

```javascript name=script.js
const display = document.getElementById('display');
const keys = document.querySelector('.keys');

let expr = ''; // current expression shown to user

function updateDisplay() {
  display.textContent = expr || '0';
}

function appendDigit(d) {
  // avoid multiple leading zeros
  if (expr === '0') expr = d;
  else expr += d;
  updateDisplay();
}

function appendDecimal() {
  // Prevent adding multiple decimals in the current number segment
  const parts = expr.split(/[\+\−\×\÷]/);
  const last = parts[parts.length - 1] || '';
  if (!last.includes('.')) {
    expr = expr + (last === '' ? '0.' : '.');
  }
  updateDisplay();
}

function appendOperator(op) {
  if (!expr) return;
  const lastChar = expr.slice(-1);
  if ('+-−×÷*/'.includes(lastChar)) {
    // replace last operator with new one
    expr = expr.slice(0, -1) + op;
  } else {
    expr += op;
  }
  updateDisplay();
}

function clearAll() {
  expr = '';
  updateDisplay();
}

function backspace() {
  expr = expr.slice(0, -1);
  updateDisplay();
}

function evaluateExpression() {
  if (!expr) return;
  // replace visual operators with JS equivalents
  const safe = expr.replace(/×/g, '*').replace(/÷/g, '/').replace(/−/g, '-');
  try {
    // Use Function instead of eval for slightly cleaner sandboxing
    // Still avoid user-supplied identifiers by restricting to expression characters
    if (!/^[0-9+\-*/. ()]+$/.test(safe)) {
      throw new Error('Invalid characters');
    }
    // Evaluate
    const result = Function('"use strict"; return (' + safe + ')')();
    expr = String(result);
  } catch (e) {
    expr = 'Error';
  }
  updateDisplay();
}

// Click handling
keys.addEventListener('click', (e) => {
  const button = e.target.closest('button');
  if (!button) return;
  const action = button.dataset.action;
  const value = button.textContent.trim();

  if (action === 'digit') appendDigit(value);
  else if (action === 'decimal') appendDecimal();
  else if (action === 'operator') appendOperator(value);
  else if (action === 'clear') clearAll();
  else if (action === 'back') backspace();
  else if (action === 'equals') evaluateExpression();
});

// Keyboard support
window.addEventListener('keydown', (e) => {
  if (e.key >= '0' && e.key <= '9') {
    appendDigit(e.key);
    e.preventDefault();
  } else if (e.key === '.') {
    appendDecimal();
    e.preventDefault();
  } else if (['+', '-', '*', '/'].includes(e.key)) {
    const map = { '+': '+', '-': '−', '*': '×', '/': '÷' };
    appendOperator(map[e.key]);
    e.preventDefault();
  } else if (e.key === 'Enter') {
    evaluateExpression();
    e.preventDefault();
  } else if (e.key === 'Backspace') {
    backspace();
    e.preventDefault();
  } else if (e.key === 'Escape') {
    clearAll();
    e.preventDefault();
  }
});

// initialize
clearAll();
```
